  const [showQuestionModal, setShowQuestionModal] = useState(false);
  const [title, setTitle] = useState('');

  const [questionImage, setQuestionImage] = useState(null);

  // ?듬? ?묒꽦 ?곹깭 (?좎깮?섏슜)
  const [replyingId, setReplyingId] = useState(null);
  const [answerText, setAnswerText] = useState('');
  const [answerImage, setAnswerImage] = useState(null);
  const [uploadingAnswerImg, setUploadingAnswerImg] = useState(false);

  const router = useRouter();

  useEffect(() => {
    const userData = localStorage.getItem('user');
    if (!userData) {
      router.push('/login');
      return;
    }
    const parsedUser = JSON.parse(userData);
    setUser(parsedUser);
    fetchQnaList();
  }, []);

  // Q&A 紐⑸줉 遺덈윭?ㅺ린
  const fetchQnaList = async () => {
    setLoading(true);
    try {
      const { data, error } = await supabase
        .from('qna')
        .select('*')
        .order('created_at', { ascending: false });

      if (error) throw error;
      setQnaList(data || []);
    } catch (err) {
      console.error(err);

  // ?대?吏 ?낅줈??怨듯넻 ?⑥닔
  const uploadImageToStorage = async (file) => {
    if (!file) return null;
    const fileExt = file.name.split('.').pop();
    const fileName = `${Date.now()}_${Math.random().toString(36).substring(2, 8)}.${fileExt}`;
    const filePath = `qna/${fileName}`;

    const { error: uploadError } = await supabase.storage
      .from('qna-images')
      .upload(filePath, file);

    if (uploadError) throw uploadError;

    const { data } = supabase.storage
      .from('qna-images')
      .getPublicUrl(filePath);

    return data.publicUrl;
  };

  // ?숈깮: 吏덈Ц ?깅줉
  const handleCreateQuestion = async (e) => {
    e.preventDefault();

      return;
    }

      let imageUrl = null;
      if (questionImage) {
        imageUrl = await uploadImageToStorage(questionImage);
      }

        {
          student_id: user.id,
          title: title,
          question: question,
          question_image_url: imageUrl,

      if (error) throw error;

    } catch (err) {
      console.error(err);
      alert(`吏덈Ц ?깅줉 ?ㅽ뙣: ${err.message}`);
    } finally {
      setUploadingQuestionImg(false);
    }
  };

  // ?좎깮?? ?듬? ?깅줉
  const handleCreateAnswer = async (qnaId) => {
    if (!answerText.trim()) {
      alert('?듬? ?댁슜???낅젰?댁＜?몄슂.');
      return;
    }

    setUploadingAnswerImg(true);

    try {
      let imageUrl = null;
      if (answerImage) {
        imageUrl = await uploadImageToStorage(answerImage);
      }

      const { error } = await supabase
        .from('qna')
        .update({
          answer: answerText,
          answer_image_url: imageUrl || undefined,
          teacher_id: user.id,
          status: 'ANSWERED',
          answered_at: new Date().toISOString(),
        })
        .eq('id', qnaId);
